\newcount\writelinecount
\newcommand{\writespace}[1]{%
  \par\vspace{0.3em}%
  \writelinecount=0
  \loop\ifnum\writelinecount<#1
    \vspace{0.9em}\noindent\mbox{}\par
    \advance\writelinecount by 1
  \repeat
  \vspace{0.8em}}

  \filbreak

  \writespace{4}}
